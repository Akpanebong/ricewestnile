DJANGO FORMS + FORMSETS (CORE) ### 📌 forms/accounting.py ```python id="f1a92" from django import forms from django.forms import inlineformset_factory from decimal import Decimal class AccountingFormForm(forms.ModelForm): class Meta: model = AccountingForm fields = ["description", "date_of_return"] class AccountingItemForm(forms.ModelForm): class Meta: model = AccountingItem fields = [ "activity_code", "program_code", "details", "amount_received", "amount_spent", ] def clean(self): cleaned = super().clean() r = cleaned.get("amount_received") or Decimal(0) s = cleaned.get("amount_spent") or Decimal(0) if s > r: raise forms.ValidationError("Spent cannot exceed received") return cleaned AccountingItemFormSet = inlineformset_factory( AccountingForm, AccountingItem, form=AccountingItemForm, extra=1, can_delete=True ) ``` --- # ⚙️ 3.
